### 6.4 标注质量检查

import os
import json
from collections import Counter

annotation_dir = "project_data/annotations/"
json_files = [f for f in os.listdir(annotation_dir) if f.endswith('.json')]

print(f"总共标注了 {len(json_files)} 张图片\n")

all_labels = []
for json_file in json_files:
    with open(os.path.join(annotation_dir, json_file)) as f:
        data = json.load(f)
        for shape in data['shapes']:
            all_labels.append(shape['label'])

label_counts = Counter(all_labels)
print("类别分布：")
for label, count in label_counts.items():
    print(f"  {label}: {count} 个实例")

# 检查类别名称一致性
unique_labels = set(all_labels)
print(f"\n检测到的类别名称：{unique_labels}")

# 推荐的类别列表
recommended = ["background", "crack"]  # 根据项目修改
extra_labels = unique_labels - set(recommended)
if extra_labels:
    print(f"⚠️ 警告：发现额外的类别名称：{extra_labels}")
    print("   请检查是否为拼写错误！")

# 检查是否有图片没有标注
print(f"\n✅ 标注检查完成")

### 8.3 数据集格式验证

**验证脚本**（`validate_dataset.py`）：

```python
import os
import numpy as np
from PIL import Image
from tqdm import tqdm

jpg_dir = "project_data/JPEGImages"
png_dir = "project_data/SegmentationClass"

jpg_files = set([f[:-4] for f in os.listdir(jpg_dir) if f.endswith('.jpg')])
png_files = set([f[:-4] for f in os.listdir(png_dir) if f.endswith('.png')])

# 检查文件名对应关系
missing_labels = jpg_files - png_files
missing_images = png_files - jpg_files

if missing_labels:
    print(f"❌ 以下图片缺少标签：{missing_labels}")
if missing_images:
    print(f"❌ 以下标签缺少图片：{missing_images}")

if not (missing_labels or missing_images):
    print("✅ 文件名对应关系检查通过")

# 检查标签像素值
print("\n检查标签像素值...")
all_pixel_values = set()

for png_file in tqdm(os.listdir(png_dir)):
    if not png_file.endswith('.png'):
        continue

    png_path = os.path.join(png_dir, png_file)
    label = np.array(Image.open(png_path))

    # 检查是否为灰度图
    if len(label.shape) > 2:
        print(f"❌ {png_file} 不是灰度图，shape={label.shape}")
        continue

    # 统计像素值
    unique_values = np.unique(label)
    all_pixel_values.update(unique_values)

print(f"\n标签中的像素值：{sorted(all_pixel_values)}")

# 验证像素值范围
num_classes = 2  # 根据实际类别数修改
expected_values = set(range(num_classes))
unexpected_values = all_pixel_values - expected_values

if unexpected_values:
    print(f"❌ 警告：发现异常像素值：{unexpected_values}")
    print(f"   预期值应为：{expected_values}")
else:
    print(f"✅ 像素值范围检查通过（{expected_values}）")
```

### 11.2 错误分析

**可视化预测错误**（`error_analysis.py`）：

```python
import os
import numpy as np
from PIL import Image
from unet import Unet
import matplotlib.pyplot as plt

# 加载最佳模型
unet = Unet()

# VOC数据集路径
VOC_path = "VOCdevkit/VOC2007"
val_list = "ImageSets/Segmentation/val.txt"

# 读取验证集
with open(os.path.join(VOC_path, val_list)) as f:
    val_ids = [line.strip() for line in f.readlines()]

# 计算每张图片的IoU
image_ious = []

for img_id in val_ids:
    # 读取图片和标签
    img_path = os.path.join(VOC_path, f"JPEGImages/{img_id}.jpg")
    gt_path = os.path.join(VOC_path, f"SegmentationClass/{img_id}.png")

    image = Image.open(img_path)
    gt = np.array(Image.open(gt_path))

    # 预测
    pred = np.array(unet.get_miou_png(image))

    # 计算IoU（简化版，仅考虑前景）
    intersection = np.sum((pred == 1) & (gt == 1))
    union = np.sum((pred == 1) | (gt == 1))
    iou = intersection / union if union > 0 else 0

    image_ious.append((img_id, iou))

# 找出IoU最低的10张图片
worst_images = sorted(image_ious, key=lambda x: x[1])[:10]

print("IoU最低的10张图片：")
for img_id, iou in worst_images:
    print(f"{img_id}.jpg - IoU: {iou:.4f}")

# 可视化错误案例
fig, axes = plt.subplots(5, 3, figsize=(15, 20))

for idx, (img_id, iou) in enumerate(worst_images[:5]):
    img_path = os.path.join(VOC_path, f"JPEGImages/{img_id}.jpg")
    gt_path = os.path.join(VOC_path, f"SegmentationClass/{img_id}.png")

    image = Image.open(img_path)
    gt = np.array(Image.open(gt_path))
    pred = np.array(unet.get_miou_png(image))

    # 显示原图、真实标签、预测结果
    axes[idx, 0].imshow(image)
    axes[idx, 0].set_title(f"{img_id} - Original")
    axes[idx, 0].axis('off')

    axes[idx, 1].imshow(gt, cmap='gray')
    axes[idx, 1].set_title(f"Ground Truth")
    axes[idx, 1].axis('off')

    axes[idx, 2].imshow(pred, cmap='gray')
    axes[idx, 2].set_title(f"Prediction (IoU: {iou:.3f})")
    axes[idx, 2].axis('off')

plt.tight_layout()
plt.savefig("error_analysis.png", dpi=150)
print("\n错误分析图已保存到 error_analysis.png")
```

### 12.3 推理脚本（简化版）

**simple_predict.py**：

```python
#!/usr/bin/env python3
"""
简化的推理脚本
用法：python simple_predict.py input.jpg output.jpg
"""

import sys
from PIL import Image
from unet import Unet

def main():
    if len(sys.argv) != 3:
        print("用法: python simple_predict.py <输入图片> <输出图片>")
        sys.exit(1)

    input_path = sys.argv[1]
    output_path = sys.argv[2]

    # 加载模型
    print("加载模型...")
    unet = Unet()

    # 读取图片
    print(f"读取图片: {input_path}")
    image = Image.open(input_path)

    # 推理
    print("推理中...")
    result = unet.detect_image(image)

    # 保存结果
    result.save(output_path)
    print(f"结果已保存到: {output_path}")

if __name__ == "__main__":
    main()
```
