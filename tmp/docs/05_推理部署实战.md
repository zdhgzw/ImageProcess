### 3.3 批量预测脚本优化

**自定义批量预测脚本**（`batch_predict.py`）：

```python
import os
from PIL import Image
from tqdm import tqdm
from unet import Unet

if __name__ == "__main__":
    # 初始化模型
    unet = Unet()

    # 配置路径
    input_dir = "test_images/"
    output_dir = "test_results/"
    os.makedirs(output_dir, exist_ok=True)

    # 获取所有图片
    image_files = [f for f in os.listdir(input_dir)
                   if f.lower().endswith(('.jpg', '.png', '.jpeg'))]

    # 批量处理
    for img_file in tqdm(image_files, desc="Processing"):
        img_path = os.path.join(input_dir, img_file)
        image = Image.open(img_path)

        # 预测
        r_image = unet.detect_image(image)

        # 保存
        save_path = os.path.join(output_dir, img_file)
        r_image.save(save_path)

    print(f"完成！处理了 {len(image_files)} 张图片")
```

运行：
```bash
python batch_predict.py
```

## 七、推理结果后处理

### 7.1 提取分割掩码

```python
from unet import Unet
from PIL import Image
import numpy as np

# 初始化模型
unet = Unet()

# 加载图片
image = Image.open("test.jpg")

# 获取分割掩码（非可视化结果）
pr = unet.get_miou_png(image)  # 返回PIL Image，每个像素值=类别ID

# 转为numpy数组
mask = np.array(pr)

# 提取特定类别的掩码
cat_mask = (mask == 1)  # 类别1（猫）的二值掩码
dog_mask = (mask == 2)  # 类别2（狗）的二值掩码

# 保存掩码
Image.fromarray((cat_mask * 255).astype(np.uint8)).save("cat_mask.png")
```

### 7.2 计算目标面积

```python
# 计算每个类别的像素数量
unique, counts = np.unique(mask, return_counts=True)

total_pixels = mask.size
for class_id, pixel_count in zip(unique, counts):
    percentage = pixel_count / total_pixels * 100
    print(f"类别 {class_id}: {pixel_count} 像素 ({percentage:.2f}%)")
```

### 7.3 提取目标区域

```python
# 根据掩码提取原图中的目标区域
image_array = np.array(image)

# 提取猫的区域
cat_region = np.where(cat_mask[:, :, None], image_array, 0)
Image.fromarray(cat_region.astype(np.uint8)).save("cat_region.jpg")

# 或：仅保留猫，背景透明（需RGBA）
image_rgba = image.convert("RGBA")
image_array_rgba = np.array(image_rgba)
image_array_rgba[:, :, 3] = cat_mask * 255  # 设置Alpha通道
Image.fromarray(image_array_rgba).save("cat_transparent.png")
```

### 7.4 边界框提取

```python
from scipy import ndimage

# 找到每个目标的连通区域
labeled, num_features = ndimage.label(cat_mask)

# 获取每个区域的边界框
for i in range(1, num_features + 1):
    # 找到该区域的坐标
    positions = np.where(labeled == i)
    y_min, y_max = positions[0].min(), positions[0].max()
    x_min, x_max = positions[1].min(), positions[1].max()

    print(f"目标 {i}: 边界框 ({x_min}, {y_min}, {x_max}, {y_max})")

    # 裁剪该区域
    cropped = image_array[y_min:y_max, x_min:x_max]
    Image.fromarray(cropped).save(f"cat_{i}.jpg")
```

### 8.3 Web API部署（Flask示例）

**app.py**：

```python
from flask import Flask, request, send_file
from PIL import Image
import io
from unet import Unet

app = Flask(__name__)
unet = Unet()

@app.route('/predict', methods=['POST'])
def predict():
    # 接收图片
    file = request.files['image']
    image = Image.open(file.stream)

    # 推理
    result = unet.detect_image(image)

    # 返回结果
    img_io = io.BytesIO()
    result.save(img_io, 'JPEG')
    img_io.seek(0)
    return send_file(img_io, mimetype='image/jpeg')

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=8000)
```

**运行API服务**：
```bash
python app.py
```

**测试API**：

